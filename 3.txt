CREATE TABLE EMPLOYEE3 (
    E_ID INT PRIMARY KEY,
    E_NAME VARCHAR(255),
    AGE INT,
    SALARY DECIMAL(10,2)
);

SELECT * FROM EMPLOYEE3;

SELECT COUNT(E_NAME) AS NUMEMPLOYEES
FROM EMPLOYEE3;

SELECT MAX(AGE) AS MAXAGE
FROM EMPLOYEE3;

SELECT MIN(AGE) AS MINAGE
FROM EMPLOYEE3;

SELECT SALARY
FROM EMPLOYEE3
ORDER BY SALARY ASC;

SELECT SALARY
FROM EMPLOYEE3
ORDER BY SALARY DESC;

SELECT SALARY, COUNT(*) AS NUMEMPLOYEES
FROM EMPLOYEE3
GROUP BY SALARY;
